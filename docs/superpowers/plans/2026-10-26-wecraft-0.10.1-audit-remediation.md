```ts
export interface CloudSaveTransport {
  save(data: unknown, revision: number, formatVersion: number, requestId: string): Promise<CloudSaveOutcome>;
}
export type CloudSaveOutcome =
  | { status: 'saved'; revision: number }
  | { status: 'unconfirmed' }
  | { status: 'unavailable' };
export interface CloudSnapshot { data: unknown; requestId: string }
export interface CloudSessionOptions {
  transport: CloudSaveTransport;
  formatVersion: number;
  newRequestId?: () => string; // 預設 () => crypto.randomUUID()
}
export class CloudSession {
  constructor(opts: CloudSessionOptions);
  get revision(): number | null;
  get pending(): CloudSnapshot | null;   // 凍結的快照（含原 requestId）
  get busy(): boolean;
  get retryable(): boolean;
  get failureKind(): CloudErrorKind | null;
  get conflict(): CloudConflict | null;  // Task 3 填入
  setRevision(revision: number | null): void;
  async checkpoint(data: unknown): Promise<void>; // 已有 pending → 重送舊快照；否則建新快照送出
  async retry(): Promise<void>;                   // 重送 pending（原 requestId）
}
```
規則：`checkpoint` 遇既有 `pending` 時**不得**換 requestId（先重送原快照）；`NETWORK`/`UNAVAILABLE` 保留 pending＋`retryable=true`；`FORMAT`/`REQUEST_REUSED`/`AUTH` 清 pending 並鎖。同時把 `src/platform/playroom.ts` 的 `save()` 改為檢查 `res.saved === true`：true → `{status:'saved', revision: res.revision}`；res 為 null → `{status:'unavailable'}`；其他 → `{status:'unconfirmed'}`。
